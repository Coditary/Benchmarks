cmake_minimum_required(VERSION 3.16)
project(compression_snappy LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(bench main.cpp)
target_include_directories(bench PRIVATE "../../../../../tools/cpp/bench-support/include")
target_compile_options(bench PRIVATE -O3 -DNDEBUG)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SNAPPY REQUIRED snappy)
target_link_libraries(bench PRIVATE snappy z)
